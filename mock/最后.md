# A区：File Directory Traversing and Mapping 超级押题

## 通用 datatype

```haskell
data Dir = File String String
         | SubDir String [Dir]
         deriving (Show, Eq)
```

---

## A1. map 所有文件内容

```haskell
mapContents :: (String -> String) -> Dir -> Dir
mapContents f (File name contents) =
  File name (f contents)

mapContents f (SubDir name dirs) =
  SubDir name (map (mapContents f) dirs)
```

---

## A2. map 所有文件名

```haskell
mapFileNames :: (String -> String) -> Dir -> Dir
mapFileNames f (File name contents) =
  File (f name) contents

mapFileNames f (SubDir name dirs) =
  SubDir name (map (mapFileNames f) dirs)
```

---

## A3. map 所有目录名

```haskell
mapDirNames :: (String -> String) -> Dir -> Dir
mapDirNames f (File name contents) =
  File name contents

mapDirNames f (SubDir name dirs) =
  SubDir (f name) (map (mapDirNames f) dirs)
```

## A10. 查找某个文件内容 Maybe

```haskell
findFile :: String -> Dir -> Maybe String
findFile target (File name contents)
  | target == name = Just contents
  | otherwise      = Nothing

findFile target (SubDir _ dirs) =
  firstJust (map (findFile target) dirs)

firstJust :: [Maybe a] -> Maybe a
firstJust [] = Nothing
firstJust (Nothing:xs) = firstJust xs
firstJust (Just x:_) = Just x
```

---

## A11. 判断文件是否存在

```haskell
fileExists :: String -> Dir -> Bool
fileExists target dir =
  case findFile target dir of
    Nothing -> False
    Just _  -> True
```

## A12. 按路径查找文件或目录

```haskell
type Path = [String]

getAtPath :: Path -> Dir -> Maybe Dir
getAtPath [] dir =
  Just dir

getAtPath (p:ps) (File _ _) =
  Nothing

getAtPath (p:ps) (SubDir _ dirs) =
  case findChild p dirs of
    Nothing -> Nothing
    Just d  -> getAtPath ps d

findChild :: String -> [Dir] -> Maybe Dir
findChild _ [] = Nothing
findChild target (d:ds)
  | dirName d == target = Just d
  | otherwise           = findChild target ds

dirName :: Dir -> String
dirName (File name _) = name
dirName (SubDir name _) = name
```

```haskell
allPaths :: Dir -> [Path]

allPaths (SubDir name dirs) =
  [name] : [name : p | d <- dirs, p <- allPaths d]
```

## A17. Maybe 删除指定文件

```haskell
deleteFile :: String -> Dir -> Maybe Dir
deleteFile target (File name contents)
  | target == name = Nothing
  | otherwise      = Just (File name contents)

deleteFile target (SubDir name dirs) =
  Just (SubDir name (deleteFromList target dirs))

deleteFromList :: String -> [Dir] -> [Dir]
deleteFromList _ [] = []
deleteFromList target (d:ds) =
  case deleteFile target d of
    Nothing -> deleteFromList target ds
    Just d' -> d' : deleteFromList target ds
```

---

## A18. 过滤文件

```haskell
filterFiles :: (String -> Bool) -> Dir -> Dir
filterFiles p (File name contents) =
  File name contents

filterFiles p (SubDir name dirs) =
  SubDir name (concatMap keep dirs)
  where
    keep (File fname contents)
      | p fname    = [File fname contents]
      | otherwise  = []
    keep d@(SubDir _ _) =
      [filterFiles p d]
```

# B区：Pruning on Trees 超级押题

## Rose Tree datatype 版本一

```haskell
data Rose a = Branch a [Rose a]
  deriving (Show, Eq)
```

---

## B1. Rose Tree 按深度 prune

保留前 `n` 层。

```haskell
pruneRose :: Int -> Rose a -> Rose a
pruneRose 0 (Branch x _) =
  Branch x []

pruneRose n (Branch x children) =
  Branch x (map (pruneRose (n - 1)) children)
```

---

## B2. Rose Tree 删除不满足条件的子树

```haskell
pruneBy :: (a -> Bool) -> Rose a -> Maybe (Rose a)
pruneBy p (Branch x children)
  | not (p x)  = Nothing
  | otherwise = Just (Branch x (mapMaybeRose p children))

mapMaybeRose :: (a -> Bool) -> [Rose a] -> [Rose a]
mapMaybeRose _ [] = []
mapMaybeRose p (t:ts) =
  case pruneBy p t of
    Nothing -> mapMaybeRose p ts
    Just t' -> t' : mapMaybeRose p ts
```

---

## B3. Rose Tree 只保留前 n 层 values

```haskell
valuesUpTo :: Int -> Rose a -> [a]
valuesUpTo 0 (Branch x _) =
  [x]

valuesUpTo n (Branch x children) =
  x : concatMap (valuesUpTo (n - 1)) children
```

---

## B4. Rose Tree 删除所有叶子

```haskell
removeLeaves :: Rose a -> Maybe (Rose a)
removeLeaves (Branch _ []) =
  Nothing

removeLeaves (Branch x children) =
  Just (Branch x (removeLeavesList children))

removeLeavesList :: [Rose a] -> [Rose a]
removeLeavesList [] = []
removeLeavesList (t:ts) =
  case removeLeaves t of
    Nothing -> removeLeavesList ts
    Just t' -> t' : removeLeavesList ts
```

---

## B5. Rose Tree 限制每个节点最多 k 个 children

```haskell
limitChildren :: Int -> Rose a -> Rose a
limitChildren k (Branch x children) =
  Branch x (map (limitChildren k) (take k children))
```

---

## Binary Tree datatype

```haskell
data BT a = Empty
          | Fork a (BT a) (BT a)
          deriving (Show, Eq)
```

---

## B6. Binary Tree 按深度 prune

```haskell
pruneBT :: Int -> BT a -> BT a
pruneBT _ Empty =
  Empty

pruneBT 0 (Fork x _ _) =
  Fork x Empty Empty

pruneBT n (Fork x l r) =
  Fork x (pruneBT (n - 1) l) (pruneBT (n - 1) r)
```

---

## B7. Binary Tree 删除不满足条件的节点

```haskell
pruneBTBy :: (a -> Bool) -> BT a -> BT a
pruneBTBy _ Empty =
  Empty

pruneBTBy p (Fork x l r)
  | p x       = Fork x (pruneBTBy p l) (pruneBTBy p r)
  | otherwise = Empty
```

---

## B8. Binary Tree collect values up to depth

```haskell
valuesBTUpTo :: Int -> BT a -> [a]
valuesBTUpTo _ Empty =
  []

valuesBTUpTo 0 (Fork x _ _) =
  [x]

valuesBTUpTo n (Fork x l r) =
  x : valuesBTUpTo (n - 1) l ++ valuesBTUpTo (n - 1) r
```

---

## B9. Binary Tree 根据 Address prune

```haskell
data Direction = L | R
  deriving (Show, Eq)

type Address = [Direction]

pruneAt :: Address -> BT a -> BT a
pruneAt [] _ =
  Empty

pruneAt _ Empty =
  Empty

pruneAt (L:ds) (Fork x l r) =
  Fork x (pruneAt ds l) r

pruneAt (R:ds) (Fork x l r) =
  Fork x l (pruneAt ds r)
```

---

## B10. Binary Tree 地址查询 Maybe

```haskell
getAtAddress :: Address -> BT a -> Maybe a
getAtAddress _ Empty =
  Nothing

getAtAddress [] (Fork x _ _) =
  Just x

getAtAddress (L:ds) (Fork _ l _) =
  getAtAddress ds l

getAtAddress (R:ds) (Fork _ _ r) =
  getAtAddress ds r
```

---

## B11. Binary Tree 返回所有合法 Address

```haskell
validAddresses :: BT a -> [Address]
validAddresses Empty =
  []

validAddresses (Fork _ l r) =
  [] :
  [L : addr | addr <- validAddresses l] ++
  [R : addr | addr <- validAddresses r]
```

---

## B12. Binary Tree 修改指定 Address 的值

```haskell
modifyAt :: Address -> (a -> a) -> BT a -> BT a
modifyAt _ _ Empty =
  Empty

modifyAt [] f (Fork x l r) =
  Fork (f x) l r

modifyAt (L:ds) f (Fork x l r) =
  Fork x (modifyAt ds f l) r

modifyAt (R:ds) f (Fork x l r) =
  Fork x l (modifyAt ds f r)
```

# C区：Frequency Tables 超级押题

## C1. 统计元素出现次数

```haskell
count :: Eq a => a -> [a] -> Int
count x xs =
  length [y | y <- xs, y == x]
```

---

## C2. 字符频率表

```haskell
freqTable :: String -> [(Char, Int)]
freqTable xs =
  [(x, count x xs) | x <- unique xs]

unique :: Eq a => [a] -> [a]
unique [] = []
unique (x:xs)
  | elem x xs = unique xs
  | otherwise = x : unique xs
```

---

## C3. 字母频率表 a-z

```haskell
import Data.Char

letterFreq :: String -> [(Char, Int)]
letterFreq xs =
  [(c, count c letters) | c <- ['a'..'z']]
  where
    letters = [toLower x | x <- xs, isAlpha x]
```

---

## C4. 百分比频率表

```haskell
percent :: Int -> Int -> Float
percent n m =
  fromIntegral n / fromIntegral m * 100

letterPercent :: String -> [(Char, Float)]
letterPercent xs =
  [(c, percent (count c letters) (length letters)) | c <- ['a'..'z']]
  where
    letters = [toLower x | x <- xs, isAlpha x]
```

---

## C5. 单词频率表

```haskell
wordFreq :: String -> [(String, Int)]
wordFreq text =
  [(w, count w ws) | w <- unique ws]
  where
    ws = words text
```

---

## C6. 按频率降序排序

```haskell
import Data.List

sortFreq :: Ord a => [(a, Int)] -> [(a, Int)]
sortFreq xs =
  sortBy compareFreq xs

compareFreq :: (a, Int) -> (a, Int) -> Ordering
compareFreq (_, n) (_, m) =
  compare m n
```

---

## C7. 找出现最多的元素

```haskell
mostCommon :: Eq a => [a] -> Maybe a
mostCommon [] =
  Nothing

mostCommon xs =
  Just (fst (maximumByFreq table))
  where
    table = [(x, count x xs) | x <- unique xs]

maximumByFreq :: [(a, Int)] -> (a, Int)
maximumByFreq [x] = x
maximumByFreq ((x,n):(y,m):xs)
  | n >= m    = maximumByFreq ((x,n):xs)
  | otherwise = maximumByFreq ((y,m):xs)
```

---

## C8. Caesar cipher 频率分析核心

```haskell
rotate :: Int -> [a] -> [a]
rotate n xs =
  drop n xs ++ take n xs

positions :: Eq a => a -> [a] -> [Int]
positions x xs =
  [i | (i,y) <- zip [0..] xs, x == y]
```

---

## C9. Chi-square 预测题

```haskell
chisqr :: [Float] -> [Float] -> Float
chisqr os es =
  sum [((o - e) ^ 2) / e | (o,e) <- zip os es]
```

---

## C10. 用 foldr 写 frequency table

```haskell
freqFold :: Eq a => [a] -> [(a, Int)]
freqFold =
  foldr insertFreq []

insertFreq :: Eq a => a -> [(a, Int)] -> [(a, Int)]
insertFreq x [] =
  [(x,1)]

insertFreq x ((y,n):ys)
  | x == y    = (y,n+1) : ys
  | otherwise = (y,n) : insertFreq x ys
```
